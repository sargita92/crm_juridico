```go
// internal/product/domain/product.go
package domain

import (
	"strings"
	"time"
)

const MaxProductNameLength = 255

type Product struct {
	ID          string
	TenantID    string
	Name        string
	Description string
	Keywords    []string
	Active      bool
	CreatedAt   time.Time
	UpdatedAt   time.Time
}

func NewProduct(id, tenantID, name, description string, keywords []string) (*Product, error) {
	if tenantID == "" {
		return nil, ErrTenantIDRequired
	}
	if name == "" {
		return nil, ErrProductNameRequired
	}
	if len(name) > MaxProductNameLength {
		return nil, ErrProductNameTooLong
	}
	if keywords == nil {
		keywords = []string{}
	}
	now := time.Now()
	return &Product{
		ID: id, TenantID: tenantID, Name: name,
		Description: description, Keywords: keywords,
		Active: true, CreatedAt: now, UpdatedAt: now,
	}, nil
}

func (p *Product) Update(name, description string, keywords []string) error {
	if name == "" {
		return ErrProductNameRequired
	}
	if len(name) > MaxProductNameLength {
		return ErrProductNameTooLong
	}
	if keywords == nil {
		keywords = []string{}
	}
	p.Name = name
	p.Description = description
	p.Keywords = keywords
	p.UpdatedAt = time.Now()
	return nil
}

func (p *Product) Activate() {
	p.Active = true
	p.UpdatedAt = time.Now()
}

func (p *Product) Deactivate() {
	p.Active = false
	p.UpdatedAt = time.Now()
}

// MatchesText checks if any keyword appears in the text (case-insensitive).
func (p *Product) MatchesText(text string) bool {
	lower := strings.ToLower(text)
	for _, kw := range p.Keywords {
		if strings.Contains(lower, strings.ToLower(kw)) {
			return true
		}
	}
	return false
}
```

```go
func (r *GormProductRepository) FindByID(ctx context.Context, id string) (*domain.Product, error) {
	var model productModel
	if err := r.db.WithContext(ctx).Where("id = ?", id).First(&model).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, domain.ErrProductNotFound
		}
		return nil, err
	}
	return productToDomain(&model), nil
}

func (r *GormProductRepository) Update(ctx context.Context, product *domain.Product) error {
	model := productToModel(product)
	result := r.db.WithContext(ctx).Save(model)
	if result.Error != nil {
		return result.Error
	}
	if result.RowsAffected == 0 {
		return domain.ErrProductNotFound
	}
	return nil
}

func (r *GormProductRepository) FindByTenantID(ctx context.Context, tenantID string, activeOnly bool) ([]domain.Product, error) {
	var models []productModel
	query := r.db.WithContext(ctx).Where("tenant_id = ?", tenantID)
	if activeOnly {
		query = query.Where("active = ?", true)
	}
	if err := query.Order("name ASC").Find(&models).Error; err != nil {
		return nil, err
	}
	products := make([]domain.Product, len(models))
	for i := range models {
		products[i] = *productToDomain(&models[i])
	}
	return products, nil
}
```

- In `Execute`, after checking lead exists:
  1. If `productDetector != nil` and `input.MessageText != ""`, call `DetectFromMessage`
  2. If product found and `funnelProductRouter != nil`, call `FindTopPriorityFunnelID`
  3. If funnel found, use that funnel instead of default; set lead's ProductID
  4. If no product or no funnel mapping, fall back to default funnel (existing behavior)

```go
func (uc *CreateLeadUseCase) Execute(ctx context.Context, input CreateLeadInput) error {
	// Check if lead already exists
	_, err := uc.leadRepo.FindByContactAndTenant(ctx, input.TenantID, input.ContactID)
	if err == nil {
		return nil
	}
	if !errors.Is(err, domain.ErrLeadNotFound) {
		return err
	}

	// Detect product from message
	var productID string
	var targetFunnel *domain.Funnel

	if uc.productDetector != nil && input.MessageText != "" {
		if detectedID, found, err := uc.productDetector.DetectFromMessage(ctx, input.TenantID, input.MessageText); err == nil && found {
			productID = detectedID
			// Try to find funnel by product priority
			if uc.funnelProductRouter != nil {
				if funnelID, err := uc.funnelProductRouter.FindTopPriorityFunnelID(ctx, detectedID); err == nil {
					if f, err := uc.funnelRepo.FindByID(ctx, funnelID); err == nil && f.Active {
						targetFunnel = f
					}
				}
			}
		}
	}

	// Fallback to default funnel
	if targetFunnel == nil {
		targetFunnel, err = uc.funnelRepo.FindDefaultByTenantID(ctx, input.TenantID)
		if err != nil {
			return err
		}
	}

	// Find entry column
	entryCol, err := uc.columnRepo.FindEntryByFunnelID(ctx, targetFunnel.ID)
	if err != nil {
		return err
	}

	// Create lead
	lead, err := domain.NewLead(uuid.New().String(), input.TenantID, targetFunnel.ID, entryCol.ID, input.ContactID, input.ConversationID)
	if err != nil {
		return err
	}
	if productID != "" {
		lead.SetProduct(productID)
	}
	if err := uc.leadRepo.Create(ctx, lead); err != nil {
		return err
	}

	// Create initial movement
	movement := domain.NewLeadMovement(uuid.New().String(), lead.ID, "", entryCol.ID)
	return uc.movementRepo.Create(ctx, movement)
}
```
