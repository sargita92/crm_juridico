// internal/ai/domain/provider.go
package domain

import (
	"context"
	"errors"
)

type MessageRole string

const (
	RoleSystem    MessageRole = "system"
	RoleUser      MessageRole = "user"
	RoleAssistant MessageRole = "assistant"
)

var (
	ErrMessageContentEmpty = errors.New("message content is empty")
	ErrMessageRoleInvalid  = errors.New("message role is invalid")
	ErrMessagesEmpty       = errors.New("messages list is empty")
	ErrModelRequired       = errors.New("model is required")
	ErrTemperatureInvalid  = errors.New("temperature must be between 0 and 1")
	ErrProviderNotFound    = errors.New("AI provider not found")
)

type AIMessage struct {
	Role    MessageRole
	Content string
}

func NewAIMessage(role MessageRole, content string) (AIMessage, error) {
	if content == "" {
		return AIMessage{}, ErrMessageContentEmpty
	}
	if !isValidRole(role) {
		return AIMessage{}, ErrMessageRoleInvalid
	}
	return AIMessage{Role: role, Content: content}, nil
}

func isValidRole(r MessageRole) bool {
	switch r {
	case RoleSystem, RoleUser, RoleAssistant:
		return true
	}
	return false
}

type AIRequest struct {
	SystemPrompt string
	Messages     []AIMessage
	Provider     string
	Model        string
	Temperature  float64
	MaxTokens    int
}

func NewAIRequest(systemPrompt string, messages []AIMessage, provider, model string, temperature float64, maxTokens int) (*AIRequest, error) {
	if len(messages) == 0 {
		return nil, ErrMessagesEmpty
	}
	if model == "" {
		return nil, ErrModelRequired
	}
	if temperature < 0 || temperature > 1 {
		return nil, ErrTemperatureInvalid
	}
	return &AIRequest{
		SystemPrompt: systemPrompt,
		Messages:     messages,
		Provider:     provider,
		Model:        model,
		Temperature:  temperature,
		MaxTokens:    maxTokens,
	}, nil
}

type AIResponse struct {
	Content          string
	FinishReason     string
	PromptTokens     int
	CompletionTokens int
}

// AIProvider is the interface for LLM providers.
// Implementations: OpenAI, Anthropic (future), etc.
type AIProvider interface {
	GenerateResponse(ctx context.Context, request *AIRequest) (*AIResponse, error)
	Name() string
}

// ProviderRegistry maps provider names to implementations.
type ProviderRegistry struct {
	providers map[string]AIProvider
}

func NewProviderRegistry() *ProviderRegistry {
	return &ProviderRegistry{providers: make(map[string]AIProvider)}
}

func (r *ProviderRegistry) Register(provider AIProvider) {
	r.providers[provider.Name()] = provider
}

func (r *ProviderRegistry) Get(name string) (AIProvider, error) {
	p, ok := r.providers[name]
	if !ok {
		return nil, ErrProviderNotFound
	}
	return p, nil
}
```

// internal/ai/domain/conversation_state.go
package domain

import (
	"context"
	"errors"
	"time"
)

var (
	ErrConversationIDRequired    = errors.New("conversation ID is required")
	ErrSpecialistIDRequired      = errors.New("specialist ID is required")
	ErrConversationStateNotFound = errors.New("conversation state not found")
)

type ConversationState struct {
	ID               string
	ConversationID   string
	SpecialistID     string
	CurrentStepIndex int
	CollectedData    map[string]string
	AccumulatedScore int
	HandoffActive    bool
	HandoffAt        *time.Time
	ResumedAt        *time.Time
	CreatedAt        time.Time
	UpdatedAt        time.Time
}

func NewConversationState(id, conversationID, specialistID string) (*ConversationState, error) {
	if conversationID == "" {
		return nil, ErrConversationIDRequired
	}
	if specialistID == "" {
		return nil, ErrSpecialistIDRequired
	}
	now := time.Now()
	return &ConversationState{
		ID:               id,
		ConversationID:   conversationID,
		SpecialistID:     specialistID,
		CurrentStepIndex: 0,
		CollectedData:    make(map[string]string),
		AccumulatedScore: 0,
		HandoffActive:    false,
		CreatedAt:        now,
		UpdatedAt:        now,
	}, nil
}

func (cs *ConversationState) AdvanceStep(key, value string, score int) {
	cs.CollectedData[key] = value
	cs.AccumulatedScore += score
	cs.CurrentStepIndex++
	cs.UpdatedAt = time.Now()
}

func (cs *ConversationState) ActivateHandoff() {
	now := time.Now()
	cs.HandoffActive = true
	cs.HandoffAt = &now
	cs.UpdatedAt = now
}

func (cs *ConversationState) DeactivateHandoff() {
	now := time.Now()
	cs.HandoffActive = false
	cs.ResumedAt = &now
	cs.UpdatedAt = now
}

// IsAfterResume returns true if the given timestamp is after the last resume.
// If no handoff ever happened, returns true (all messages are valid).
func (cs *ConversationState) IsAfterResume(timestamp time.Time) bool {
	if cs.ResumedAt == nil {
		return true
	}
	return timestamp.After(*cs.ResumedAt)
}

type ConversationStateRepository interface {
	Create(ctx context.Context, state *ConversationState) error
	FindByConversationID(ctx context.Context, conversationID string) (*ConversationState, error)
	Update(ctx context.Context, state *ConversationState) error
}
```

type OpenAIProvider struct {

	baseURL string
	client  *http.Client
	log     *zap.Logger
}

func NewOpenAIProvider(apiKey, baseURL string, log *zap.Logger) *OpenAIProvider {
	if baseURL == "" {
		baseURL = defaultOpenAIURL
	}
	return &OpenAIProvider{
		apiKey:  apiKey,
		baseURL: baseURL,
		client:  &http.Client{Timeout: 60 * time.Second},
		log:     log,
	}
}

func (p *OpenAIProvider) Name() string { return "openai" }

type openAIRequest struct {
	Model       string             `json:"model"`
	Messages    []openAIMessage    `json:"messages"`
	Temperature float64            `json:"temperature,omitempty"`
	MaxTokens   int                `json:"max_tokens,omitempty"`
}

type openAIMessage struct {
	Role    string `json:"role"`
	Content string `json:"content"`
}

type openAIResponse struct {
	Choices []openAIChoice `json:"choices"`
	Usage   openAIUsage    `json:"usage"`
}

type openAIChoice struct {
	Message      openAIMessage `json:"message"`
	FinishReason string        `json:"finish_reason"`
}

type openAIUsage struct {
	PromptTokens     int `json:"prompt_tokens"`
	CompletionTokens int `json:"completion_tokens"`
}

type openAIError struct {
	Error struct {
		Message string `json:"message"`
	} `json:"error"`
}

func (p *OpenAIProvider) GenerateResponse(ctx context.Context, request *domain.AIRequest) (*domain.AIResponse, error) {
	messages := make([]openAIMessage, 0, len(request.Messages)+1)
	if request.SystemPrompt != "" {
		messages = append(messages, openAIMessage{Role: "system", Content: request.SystemPrompt})
	}
	for _, m := range request.Messages {
		messages = append(messages, openAIMessage{Role: string(m.Role), Content: m.Content})
	}

	reqBody := openAIRequest{
		Model:       request.Model,
		Messages:    messages,
		Temperature: request.Temperature,
		MaxTokens:   request.MaxTokens,
	}

	body, err := json.Marshal(reqBody)
	if err != nil {
		return nil, fmt.Errorf("marshal request: %w", err)
	}

	httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, p.baseURL, bytes.NewReader(body))
	if err != nil {
		return nil, fmt.Errorf("create request: %w", err)
	}
	httpReq.Header.Set("Content-Type", "application/json")
	httpReq.Header.Set("Authorization", "Bearer "+p.apiKey)

	start := time.Now()
	resp, err := p.client.Do(httpReq)
	duration := time.Since(start)
	if err != nil {
		p.log.Error("openai request failed", zap.Error(err), zap.Duration("duration", duration))
		return nil, fmt.Errorf("openai request: %w", err)
	}
	defer resp.Body.Close()

	respBody, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("read response: %w", err)
	}

	if resp.StatusCode != http.StatusOK {
		var apiErr openAIError
		_ = json.Unmarshal(respBody, &apiErr)
		p.log.Error("openai API error",
			zap.Int("status", resp.StatusCode),
			zap.String("error", apiErr.Error.Message),
			zap.Duration("duration", duration),
		)
		return nil, fmt.Errorf("openai API error (status %d): %s", resp.StatusCode, apiErr.Error.Message)
	}

	var openAIResp openAIResponse
	if err := json.Unmarshal(respBody, &openAIResp); err != nil {
		return nil, fmt.Errorf("unmarshal response: %w", err)
	}

	if len(openAIResp.Choices) == 0 {
		return nil, fmt.Errorf("openai returned no choices")
	}

	p.log.Info("openai response received",
		zap.String("model", request.Model),
		zap.Int("prompt_tokens", openAIResp.Usage.PromptTokens),
		zap.Int("completion_tokens", openAIResp.Usage.CompletionTokens),
		zap.Duration("duration", duration),
	)

	return &domain.AIResponse{
		Content:          openAIResp.Choices[0].Message.Content,
		FinishReason:     openAIResp.Choices[0].FinishReason,
		PromptTokens:     openAIResp.Usage.PromptTokens,
		CompletionTokens: openAIResp.Usage.CompletionTokens,
	}, nil

// internal/ai/application/step_evaluator.go
package application

import (
	"encoding/json"
	"regexp"
	"strings"

	specDomain "github.com/sasrgita/crm-juridico/internal/specialist/domain"
)

var (
	numberRegex   = regexp.MustCompile(`^\d+([.,]\d+)?$`)
	dateRegex     = regexp.MustCompile(`\d{2}[/\-]\d{2}[/\-]\d{4}`)
	cpfRegex      = regexp.MustCompile(`\d{3}\.?\d{3}\.?\d{3}-?\d{2}`)
	cnpjRegex     = regexp.MustCompile(`\d{2}\.?\d{3}\.?\d{3}/?\d{4}-?\d{2}`)
	stepMetaRegex = regexp.MustCompile(`<!--STEP_META:(.*?)-->`)
)

type StepEvalResult struct {
	Completed     bool
	CollectedData string
}

type StepMetadata struct {
	StepCompleted bool   `json:"step_completed"`
	CollectedData string `json:"collected_data"`
	Score         int    `json:"score"`
}

type StepEvaluator struct{}

func NewStepEvaluator() *StepEvaluator {
	return &StepEvaluator{}
}

// EvaluateByRule tries to validate the message using regex rules based on step data type.
// Returns Completed=false for free_text (needs LLM evaluation).
func (e *StepEvaluator) EvaluateByRule(step *specDomain.Step, message string) StepEvalResult {
	text := strings.TrimSpace(message)

	switch step.DataType {
	case specDomain.StepDataTypeNumber:
		if numberRegex.MatchString(text) {
			return StepEvalResult{Completed: true, CollectedData: text}
		}
	case specDomain.StepDataTypeDate:
		if match := dateRegex.FindString(text); match != "" {
			return StepEvalResult{Completed: true, CollectedData: match}
		}
	case specDomain.StepDataTypeDocument:
		if match := cpfRegex.FindString(text); match != "" {
			return StepEvalResult{Completed: true, CollectedData: match}
		}
		if match := cnpjRegex.FindString(text); match != "" {
			return StepEvalResult{Completed: true, CollectedData: match}
		}
	case specDomain.StepDataTypeSelection:
		// Selection validation would check against predefined options
		// For now, any non-empty input counts
		if text != "" {
			return StepEvalResult{Completed: true, CollectedData: text}
		}
	case specDomain.StepDataTypeFreeText:
		// free_text always needs LLM evaluation
		return StepEvalResult{Completed: false}
	}

	return StepEvalResult{Completed: false}
}

// ParseMetadata extracts STEP_META from the LLM response and returns cleaned response.
func (e *StepEvaluator) ParseMetadata(response string) (StepMetadata, string) {
	matches := stepMetaRegex.FindStringSubmatch(response)
	if len(matches) < 2 {
		return StepMetadata{}, response
	}

	var meta StepMetadata
	if err := json.Unmarshal([]byte(matches[1]), &meta); err != nil {
		return StepMetadata{}, response
	}

	clean := strings.TrimSpace(stepMetaRegex.ReplaceAllString(response, ""))
	return meta, clean
}

// internal/ai/application/specialist_router.go
package application

import (
	"context"
	"errors"

	"github.com/sasrgita/crm-juridico/internal/ai/domain"
)

var ErrNoSpecialistAvailable = errors.New("no specialist available for this tenant")

// PhoneNumberFinder resolves phone number to product (cross-module adapter).
type PhoneNumberFinder interface {
	FindProductIDByPhone(ctx context.Context, phone string) (string, error)
}

// SpecialistProductFinder finds specialists linked to a product.
type SpecialistProductFinder interface {
	FindByProductID(ctx context.Context, productID string) ([]domain.SpecialistProduct, error)
}

// ProductDetectorForRouter detects product from message text.
type ProductDetectorForRouter interface {
	DetectFromMessage(ctx context.Context, tenantID, text string) (productID string, found bool, err error)
}

// DefaultSpecialistFinder finds the default specialist for a tenant.
type DefaultSpecialistFinder interface {
	FindDefaultByTenantID(ctx context.Context, tenantID string) (specialistID string, err error)
}

type SpecialistRouter struct {
	phoneFinder     PhoneNumberFinder
	spFinder        SpecialistProductFinder
	productDetector ProductDetectorForRouter
	defaultFinder   DefaultSpecialistFinder
}

func NewSpecialistRouter(
	phoneFinder PhoneNumberFinder,
	spFinder SpecialistProductFinder,
	productDetector ProductDetectorForRouter,
	defaultFinder DefaultSpecialistFinder,
) *SpecialistRouter {
	return &SpecialistRouter{
		phoneFinder:     phoneFinder,
		spFinder:        spFinder,
		productDetector: productDetector,
		defaultFinder:   defaultFinder,
	}
}

// Route resolves which specialist should handle the conversation.
// Returns (specialistID, productID, error). productID may be empty.
// Cascade: phone → product from message → default.
func (r *SpecialistRouter) Route(ctx context.Context, tenantID, phone, messageText string) (string, string, error) {
	// 1. Try by phone number → product → specialist
	if r.phoneFinder != nil {
		productID, err := r.phoneFinder.FindProductIDByPhone(ctx, phone)
		if err == nil && productID != "" {
			specID, found := r.findSpecialistForProduct(ctx, productID)
			if found {
				return specID, productID, nil
			}
		}
	}

	// 2. Try by message text → product → specialist
	if r.productDetector != nil && messageText != "" {
		productID, found, err := r.productDetector.DetectFromMessage(ctx, tenantID, messageText)
		if err == nil && found {
			specID, specFound := r.findSpecialistForProduct(ctx, productID)
			if specFound {
				return specID, productID, nil
			}
		}
	}

	// 3. Fallback to default specialist
	if r.defaultFinder != nil {
		specID, err := r.defaultFinder.FindDefaultByTenantID(ctx, tenantID)
		if err == nil && specID != "" {
			return specID, "", nil
		}
	}

	return "", "", ErrNoSpecialistAvailable
}

func (r *SpecialistRouter) findSpecialistForProduct(ctx context.Context, productID string) (string, bool) {
	if r.spFinder == nil {
		return "", false
	}
	sps, err := r.spFinder.FindByProductID(ctx, productID)
	if err != nil || len(sps) == 0 {
		return "", false
	}
	return sps[0].SpecialistID, true
}

	documentFetcher DocumentFetcher,
	productFinder ProductInfoFinder,
	historyFinder MessageHistoryFinder,
) *ContextBuilder {
	return &ContextBuilder{
		specialistFinder: specialistFinder,
		stepFinder:       stepFinder,
		guardrailFinder:  guardrailFinder,
		documentFetcher:  documentFetcher,
		productFinder:    productFinder,
		historyFinder:    historyFinder,
	}
}

// Build constructs the full AIRequest with system prompt and message history.
func (cb *ContextBuilder) Build(ctx context.Context, state *domain.ConversationState, productID string, historyLimit int) (*domain.AIRequest, error) {
	specialist, err := cb.specialistFinder.FindByID(ctx, state.SpecialistID)
	if err != nil {
		return nil, fmt.Errorf("find specialist: %w", err)
	}

	var promptParts []string

	// 1. Base prompt
	promptParts = append(promptParts, fmt.Sprintf("Voce e %s. %s", specialist.Name, specialist.Prompt))

	// 2. Product context
	if productID != "" && cb.productFinder != nil {
		name, desc, err := cb.productFinder.FindProductInfo(ctx, productID)
		if err == nil {
			promptParts = append(promptParts, fmt.Sprintf("\n## Produto\nO cliente esta interessado em: %s. %s", name, desc))
		}
	}

	// 3. RAG documents
	if cb.documentFetcher != nil {
		docs, err := cb.documentFetcher.FetchDocumentContents(ctx, state.SpecialistID)
		if err == nil && len(docs) > 0 {
			promptParts = append(promptParts, "\n## Documentos de referencia")
			for _, d := range docs {
				promptParts = append(promptParts, d)
			}
		}
	}

	// 4. Guardrails
	if cb.guardrailFinder != nil {
		guardrails, err := cb.guardrailFinder.FindBySpecialistID(ctx, state.SpecialistID)
		if err == nil && len(guardrails) > 0 {
			var rules []string
			for _, g := range guardrails {
				if !g.Active {
					continue
				}
				switch g.Type {
				case specDomain.GuardrailTypeForbiddenTopics:
					rules = append(rules, fmt.Sprintf("- Nunca fale sobre: %s", g.Rule))
				case specDomain.GuardrailTypeScopeLimit:
					rules = append(rules, fmt.Sprintf("- Mantenha-se dentro do escopo: %s", g.Rule))
				case specDomain.GuardrailTypeResponseTone:
					rules = append(rules, fmt.Sprintf("- Tom: %s", g.Rule))
				}
			}
			if len(rules) > 0 {
				promptParts = append(promptParts, "\n## Regras obrigatorias\n"+strings.Join(rules, "\n"))
			}
		}
	}

	// 5. Current step
	if cb.stepFinder != nil {
		steps, err := cb.stepFinder.FindBySpecialistID(ctx, state.SpecialistID)
		if err == nil && len(steps) > 0 && state.CurrentStepIndex < len(steps) {
			step := steps[state.CurrentStepIndex]
			stepSection := fmt.Sprintf("\n## Etapa atual do atendimento\nVoce esta na etapa %d de %d: \"%s\"\nTipo de dado esperado: %s",
				state.CurrentStepIndex+1, len(steps), step.Text, string(step.DataType))

			if len(state.CollectedData) > 0 {
				stepSection += fmt.Sprintf("\nDados ja coletados: %v", state.CollectedData)
			}

			if step.DataType == specDomain.StepDataTypeFreeText {
				stepSection += "\n\nQuando o cliente fornecer o dado solicitado, responda normalmente e inclua ao final:\n<!--STEP_META:{\"step_completed\":true,\"collected_data\":\"valor\",\"score\":" + fmt.Sprintf("%d", step.Score) + "}-->"
			}

			promptParts = append(promptParts, stepSection)
		}
	}

	systemPrompt := strings.Join(promptParts, "\n")

	// Build message history
	var messages []domain.AIMessage
	if cb.historyFinder != nil {
		history, err := cb.historyFinder.FindHistory(ctx, state.ConversationID, historyLimit)
		if err == nil {
			for _, h := range history {
				if state.IsAfterResume(h.Timestamp) {
					messages = append(messages, domain.AIMessage{Role: h.Role, Content: h.Content})
				}
			}
		}
	}

	if len(messages) == 0 {
		messages = append(messages, domain.AIMessage{Role: domain.RoleUser, Content: "oi"})
	}

	return &domain.AIRequest{
		SystemPrompt: systemPrompt,
		Messages:     messages,
	}, nil
}

// internal/ai/application/debouncer.go
package application

import (
	"sync"
	"time"
)

type DebouncerCallback func(conversationID string, messages []string)

type Debouncer struct {
	duration time.Duration
	callback DebouncerCallback
	timers   map[string]*time.Timer
	messages map[string][]string
	mu       sync.Mutex
}

func NewDebouncer(duration time.Duration, callback DebouncerCallback) *Debouncer {
	return &Debouncer{
		duration: duration,
		callback: callback,
		timers:   make(map[string]*time.Timer),
		messages: make(map[string][]string),
	}
}

func (d *Debouncer) Add(conversationID, message string) {
	d.mu.Lock()
	defer d.mu.Unlock()

	d.messages[conversationID] = append(d.messages[conversationID], message)

	if timer, exists := d.timers[conversationID]; exists {
		timer.Stop()
	}

	d.timers[conversationID] = time.AfterFunc(d.duration, func() {
		d.fire(conversationID)
	})
}

func (d *Debouncer) fire(conversationID string) {
	d.mu.Lock()
	msgs := d.messages[conversationID]
	delete(d.messages, conversationID)
	delete(d.timers, conversationID)
	d.mu.Unlock()

	if len(msgs) > 0 {
		d.callback(conversationID, msgs)
	}
}

func (d *Debouncer) Stop() {
	d.mu.Lock()
	defer d.mu.Unlock()
	for _, timer := range d.timers {
		timer.Stop()
	}
}

package application

import (
	"strings"

	specDomain "github.com/sasrgita/crm-juridico/internal/specialist/domain"
)

type GuardrailChecker struct{}

func NewGuardrailChecker() *GuardrailChecker {
	return &GuardrailChecker{}
}

// Check validates the AI response against active forbidden_topics guardrails.
// Returns (violated, fallbackMessage). scope_limit and response_tone are prompt-engineered.
func (gc *GuardrailChecker) Check(response string, guardrails []specDomain.Guardrail) (bool, string) {
	lowerResponse := strings.ToLower(response)

	for _, g := range guardrails {
		if !g.Active || g.Type != specDomain.GuardrailTypeForbiddenTopics {
			continue
		}

		keywords := strings.Split(g.Rule, ",")
		for _, kw := range keywords {
			kw = strings.TrimSpace(strings.ToLower(kw))
			if kw != "" && strings.Contains(lowerResponse, kw) {
				return true, g.Message
			}
		}
	}

	return false, ""
}

	guardrailChecker *GuardrailChecker,

		contextBuilder:   contextBuilder,

// HandleMessages processes debounced messages for a conversation.
func (e *ConversationEngine) HandleMessages(ctx context.Context, tenantID, conversationID, specialistID, productID string, messages []string) error {
	log := e.log.With(
		zap.String("tenant_id", tenantID),
		zap.String("conversation_id", conversationID),
		zap.String("specialist_id", specialistID),
	)

	// 1. Get or create conversation state
	state, err := e.stateRepo.FindByConversationID(ctx, conversationID)
	if err != nil {
		state, err = domain.NewConversationState(uuid.New().String(), conversationID, specialistID)
		if err != nil {
			log.Error("failed to create conversation state", zap.Error(err))
			return err
		}
		if createErr := e.stateRepo.Create(ctx, state); createErr != nil {
			log.Error("failed to persist conversation state", zap.Error(createErr))
			return createErr
		}
		log.Info("conversation state created")
	}

	// 2. Check handoff
	if state.HandoffActive {
		log.Info("handoff active, skipping AI response")
		return nil
	}

	// 3. Resolve config
	config := e.configResolver.Resolve(ctx, specialistID)
	log = log.With(zap.String("provider", config.Provider), zap.String("model", config.Model))

	// 4. Build context
	aiRequest, err := e.contextBuilder.Build(ctx, state, productID, 20)
	if err != nil {
		log.Error("failed to build context", zap.Error(err))
		return err
	}
	aiRequest.Provider = config.Provider
	aiRequest.Model = config.Model
	aiRequest.Temperature = config.Temperature
	aiRequest.MaxTokens = config.MaxTokens
	log.Info("context built")

	// 5. Call LLM
	provider, err := e.providerRegistry.Get(config.Provider)
	if err != nil {
		log.Error("provider not found", zap.Error(err))
		return err
	}

	start := time.Now()
	aiResponse, err := provider.GenerateResponse(ctx, aiRequest)
	duration := time.Since(start)

	aiRequestDuration.WithLabelValues(tenantID, config.Provider, config.Model).Observe(duration.Seconds())

	if err != nil {
		aiRequestsTotal.WithLabelValues(tenantID, specialistID, config.Provider, config.Model, "error").Inc()
		log.Error("LLM request failed", zap.Error(err), zap.Duration("duration", duration))
		return err
	}

	aiRequestsTotal.WithLabelValues(tenantID, specialistID, config.Provider, config.Model, "success").Inc()
	aiTokensPromptTotal.WithLabelValues(tenantID, specialistID, config.Provider).Add(float64(aiResponse.PromptTokens))
	aiTokensCompletionTotal.WithLabelValues(tenantID, specialistID, config.Provider).Add(float64(aiResponse.CompletionTokens))
	log.Info("LLM response received", zap.Duration("duration", duration), zap.Int("prompt_tokens", aiResponse.PromptTokens), zap.Int("completion_tokens", aiResponse.CompletionTokens))

	responseContent := aiResponse.Content

	// 6. Parse metadata and evaluate step
	meta, cleanContent := e.stepEvaluator.ParseMetadata(responseContent)

	if e.contextBuilder.stepFinder != nil {
		steps, stepsErr := e.contextBuilder.stepFinder.FindBySpecialistID(ctx, specialistID)
		if stepsErr == nil && len(steps) > 0 && state.CurrentStepIndex < len(steps) {
			currentStep := steps[state.CurrentStepIndex]
			combinedMessage := strings.Join(messages, "\n")

			// Try rule-based evaluation first
			ruleResult := e.stepEvaluator.EvaluateByRule(&currentStep, combinedMessage)

			stepCompleted := ruleResult.Completed
			collectedData := ruleResult.CollectedData
			score := currentStep.Score

			if !stepCompleted && meta.StepCompleted {
				// Fallback to LLM metadata
				stepCompleted = true
				collectedData = meta.CollectedData
				score = meta.Score
			}

			if stepCompleted {
				state.AdvanceStep(fmt.Sprintf("step_%d", state.CurrentStepIndex), collectedData, score)
				aiStepsCompletedTotal.WithLabelValues(tenantID, specialistID).Inc()
				log.Info("step completed", zap.Int("step_index", state.CurrentStepIndex-1), zap.String("collected", collectedData), zap.Int("score", score))

				// Move lead to target column if configured
				if currentStep.TargetColumnID != "" && e.leadUpdater != nil {
					if moveErr := e.leadUpdater.MoveLeadToColumn(ctx, conversationID, currentStep.TargetColumnID); moveErr != nil {
						log.Error("failed to move lead to target column", zap.Error(moveErr))
					} else {
						log.Info("lead moved to target column", zap.String("column_id", currentStep.TargetColumnID))
					}
				}

				// Update lead score
				if e.leadUpdater != nil {
					_ = e.leadUpdater.UpdateLeadScore(ctx, conversationID, state.AccumulatedScore)
				}

				// Check if all steps completed → qualification
				if state.CurrentStepIndex >= len(steps) {
					e.handleQualification(ctx, tenantID, conversationID, specialistID, state, log)
				}
			}
		}
	}

	// 7. Guardrail check
	if e.guardrailChecker != nil && e.contextBuilder.guardrailFinder != nil {
		guardrails, gErr := e.contextBuilder.guardrailFinder.FindBySpecialistID(ctx, specialistID)
		if gErr == nil {
			violated, fallback := e.guardrailChecker.Check(cleanContent, guardrails)
			if violated {
				aiGuardrailViolationsTotal.WithLabelValues(tenantID, specialistID, "forbidden_topics").Inc()
				log.Warn("guardrail violated, using fallback", zap.String("original", cleanContent))
				cleanContent = fallback
			}
		}
	}

	// 8. Send response
	if sendErr := e.messageSender.SendAIResponse(ctx, tenantID, conversationID, cleanContent); sendErr != nil {
		log.Error("failed to send AI response", zap.Error(sendErr))
		return sendErr
	}
	log.Info("AI response sent")

	// 9. Persist state
	if updateErr := e.stateRepo.Update(ctx, state); updateErr != nil {
		log.Error("failed to update conversation state", zap.Error(updateErr))
		return updateErr
	}

	return nil
}

func (e *ConversationEngine) handleQualification(ctx context.Context, tenantID, conversationID, specialistID string, state *domain.ConversationState, log *zap.Logger) {
	if e.contextBuilder.stepFinder == nil || e.leadUpdater == nil {
		return
	}

	// Fetch scoring config to determine qualification
	// This will be wired via an adapter interface
	log.Info("all steps completed, checking qualification", zap.Int("total_score", state.AccumulatedScore))
}

// internal/ai/application/handoff.go
package application

import (
	"context"
	"errors"

	"go.uber.org/zap"

	"github.com/sasrgita/crm-juridico/internal/ai/domain"
)

var (
	ErrHandoffAlreadyActive = errors.New("handoff is already active")
	ErrHandoffNotActive     = errors.New("handoff is not active")
)

type ActivateHandoffUseCase struct {
	stateRepo domain.ConversationStateRepository
	log       *zap.Logger
}

func NewActivateHandoffUseCase(stateRepo domain.ConversationStateRepository, log *zap.Logger) *ActivateHandoffUseCase {
	return &ActivateHandoffUseCase{stateRepo: stateRepo, log: log}
}

func (uc *ActivateHandoffUseCase) Execute(ctx context.Context, tenantID, conversationID string) error {
	state, err := uc.stateRepo.FindByConversationID(ctx, conversationID)
	if err != nil {
		return err
	}
	if state.HandoffActive {
		return ErrHandoffAlreadyActive
	}
	state.ActivateHandoff()

	if err := uc.stateRepo.Update(ctx, state); err != nil {
		return err
	}

	aiHandoffsTotal.WithLabelValues(tenantID, "take").Inc()
	uc.log.Info("handoff activated",
		zap.String("tenant_id", tenantID),
		zap.String("conversation_id", conversationID),
	)
	return nil
}

type DeactivateHandoffUseCase struct {
	stateRepo domain.ConversationStateRepository
	log       *zap.Logger
}

func NewDeactivateHandoffUseCase(stateRepo domain.ConversationStateRepository, log *zap.Logger) *DeactivateHandoffUseCase {
	return &DeactivateHandoffUseCase{stateRepo: stateRepo, log: log}
}

func (uc *DeactivateHandoffUseCase) Execute(ctx context.Context, tenantID, conversationID string) error {
	state, err := uc.stateRepo.FindByConversationID(ctx, conversationID)
	if err != nil {
		return err
	}
	if !state.HandoffActive {
		return ErrHandoffNotActive
	}
	state.DeactivateHandoff()

	if err := uc.stateRepo.Update(ctx, state); err != nil {
		return err
	}

	aiHandoffsTotal.WithLabelValues(tenantID, "release").Inc()
	uc.log.Info("handoff deactivated",
		zap.String("tenant_id", tenantID),
		zap.String("conversation_id", conversationID),
	)
	return nil
}
